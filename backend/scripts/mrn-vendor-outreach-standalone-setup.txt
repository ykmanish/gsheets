HOW TO READ THE MRN SHEET (no credentials needed)
============================================================
The MRN sheet is shared "anyone with the link can view", so it's readable via Google's CSV export endpoint with a plain HTTP fetch — no Google auth, no service account, no MCP connector required:

  https://docs.google.com/spreadsheets/d/1E2ggJmYVK26DPrKFTI0KyMV_R6HktQaxnvglGw3SjWA/gviz/tq?tqx=out:csv&sheet=MRN%20Form

Fetch it with curl (via Bash), not WebFetch — WebFetch summarizes content through a model and isn't reliable for exact structured data:
  curl -s -L "https://docs.google.com/spreadsheets/d/1E2ggJmYVK26DPrKFTI0KyMV_R6HktQaxnvglGw3SjWA/gviz/tq?tqx=out:csv&sheet=MRN%20Form" -o /tmp/mrn.csv

Then parse it into rows with this exact Node script (CSV needs a real parser here, not naive comma-splitting, because `Material Requirement` cells contain embedded commas and newlines inside quoted fields). Save this to a temp .js file and run it with `node <file> /tmp/mrn.csv`:

  const fs = require('fs');
  const text = fs.readFileSync(process.argv[2], 'utf8');
  function parseCSV(str) {
    const rows = []; let row = []; let field = ''; let inQuotes = false;
    for (let i = 0; i < str.length; i++) {
      const c = str[i];
      if (inQuotes) {
        if (c === '"') { if (str[i+1] === '"') { field += '"'; i++; } else { inQuotes = false; } }
        else field += c;
      } else {
        if (c === '"') inQuotes = true;
        else if (c === ',') { row.push(field); field=''; }
        else if (c === '\r') {}
        else if (c === '\n') { row.push(field); rows.push(row); row=[]; field=''; }
        else field += c;
      }
    }
    if (field.length || row.length) { row.push(field); rows.push(row); }
    return rows;
  }
  const rows = parseCSV(text);
  const headers = rows[0];
  const idx = name => headers.findIndex(h => h.trim().toLowerCase() === name.toLowerCase());
  const col = { mrnNo: idx('MRN No'), timestamp: idx('Timestamp'), project: headers.findIndex(h => h.trim().toUpperCase().startsWith('PROJECT NAME')), reqDate: idx('Material Request Date'), byWhen: idx('By when Material is Required'), material: idx('Material Requirement'), issuedBy: headers.findIndex(h => h.trim().toLowerCase().startsWith('issued by')), category: headers.findIndex(h => ['category','catagory'].includes(h.trim().toLowerCase())) };
  const records = [];
  for (let i = 1; i < rows.length; i++) {
    const r = rows[i];
    const mrnNo = (r[col.mrnNo] || '').trim();
    const material = (r[col.material] || '').trim();
    if (!mrnNo && !material) continue;
    records.push({
      rowNumber: i + 1,
      mrnNo: mrnNo || ('MRN' + String(i).padStart(2, '0')),
      timestamp: (r[col.timestamp] || '').trim(),
      project: (r[col.project] || '').trim(),
      materialRequestDate: (r[col.reqDate] || '').trim(),
      requiredDate: (r[col.byWhen] || '').trim(),
      materialRequirement: material,
      issuedBy: (r[col.issuedBy] || '').trim(),
      category: (r[col.category] || '').trim(),
    });
  }
  console.log(JSON.stringify({ totalRecords: records.length, maxRowNumber: rows.length, records }, null, 2));
